    const timelineItems = [

      { date: "2023-06-10", type: "疫苗", description: "接种疫苗A" },

      { date: "2023-12-01", type: "体重", description: "体重达2.5kg" },

      { date: "2024-01-15", type: "疫苗", description: "接种疫苗B" }

    const friends = [

      { id: 1, city: "北京", name: "小蓝", pet: { name: "豆豆", breed: "米花犬" }, avatar: "" },

      { id: 2, city: "上海", name: "芳芳", pet: { name: "雪球", breed: "雪纳瑞" }, avatar: "" },

      { id: 3, city: "广州", name: "乐乐", pet: { name: "跳跳", breed: "柯基" }, avatar: "" },

    // DOM helpers

    const petsListEl = document.getElementById('pets-list');

    const timelineEl = document.getElementById('timeline-items');

    const timelineFill = document.getElementById('timeline-fill');

    const friendListEl = document.getElementById('friend-list');

    const cityFilterEl = document.getElementById('city-filter');

    const feedListEl = document.getElementById('feed-list');

    const publishBtn = document.getElementById('publish-post');

    const postTitle = document.getElementById('post-title');

    const postPhoto = document.getElementById('post-photo');

    const postContent = document.getElementById('post-content');

 

    // 初始渲染

    function renderPets() {

      petsListEl.innerHTML = "";

      // 仅演示：每一个宠物一张卡

      pets.forEach(p => {

        const card = document.createElement('div');

        card.className = 'pet-card';

        card.innerHTML = `

          <div class="avatar" aria-label="宠物头像">${p.name.slice(0,1)}</div>

          <div class="pet-info">

            <div style="font-weight:600">${p.name} • ${p.breed}</div>

            <div class="muted" style="font-size:12px">生日：${p.birth}</div>

            <div class="section-sub" style="margin-top:6px; display:flex; gap:6px; flex-wrap:wrap">

              <span class="tag">健康记录：${p.health.length} 条</span>

              <span class="tag">ID: ${p.id}</span>

            </div>

            <div class="section-sub" style="margin-top:6px; display:flex; gap:6px; flex-wrap:wrap">

              ${p.health.map(h => `<span class="tag" title="${h.date}">${h.date}</span>`).join('')}

            </div>

          </div>

        `;

        petsListEl.appendChild(card);

      });

    }

 

    function renderTimeline() {

      // 将健康记录中的日期与成长时间线数据合并，排序

      const items = [];

      timelineItems.forEach(t => items.push(t));

      pets.forEach(p => {

        p.health.forEach(h => {

          items.push({ date: h.date, type: "健康记录", description: `${p.name} - ${h.weight}kg` });

        });

      });

      items.sort((a,b)=> new Date(a.date) - new Date(b.date));

      timelineEl.innerHTML = "";

      items.forEach(it => {

        const li = document.createElement('li');

        li.style.flex = "0 0 auto";

        li.innerHTML = `

          <div style="font-size:12px; color:#888">${it.date}</div>

          <div style="font-weight:600; margin-top:4px">${it.type}: ${it.description}</div>

        `;

        timelineEl.appendChild(li);

      });

      // 计算进度示意

      const total = items.length || 1;

      timelineFill.style.width = Math.min(100, total * 8) + "%";

    }

 

    function renderFriends(filterCity = "") {

      friendListEl.innerHTML = "";

      const list = friends.filter(f => filterCity ? f.city === filterCity : true);

      list.forEach(ff => {

        const card = document.createElement('div');

        card.style.display = "flex";

        card.style.alignItems = "center";

        card.style.padding = "6px 8px";

        card.style.borderRadius = "8px";

        card.style.border = "1px solid #eee";

        card.style.background = "#fff";

        card.style.minWidth = "120px";

        card.style.cursor = "default";

        card.innerHTML = `

          <div class="avatar-sm" aria-label="头像" style="background:#ddd"></div>

          <div style="font-size:12px;">

            <div><strong>${ff.name}</strong> • ${ff.city}</div>

            <div class="muted" style="font-size:11px">${ff.pet.name}（${ff.pet.breed}）</div>

          </div>

        `;

        friendListEl.appendChild(card);

      });

    }

 

    function renderFeed() {

      const posts = [

        { id: 1, author: "小蓝", title: "豆豆第一次公共场所训练成效不错", content: "今天带豆豆去公园，学会了坐下和等候，真开心！", photo: "" },

      ];

      posts.forEach(p => {

        const c = document.createElement('div');

        c.className = 'card';

        c.innerHTML = `

          <div style="display:flex; align-items:center; gap:8px;">

            <span class="avatar-sm" aria-label="avatar"></span>

            <strong>${p.author}</strong>

            <span class="muted" style="font-size:12px;">发布于：今天</span>

          <div class="muted" style="font-size:12px; margin-top:4px">${p.content}</div>

          ${p.photo ? `<img src="${p.photo}" alt="图文封面" />` : ""}

        `;

        feedListEl.appendChild(c);

      });

    }

 

    // 事件绑定

    document.getElementById('add-pet').addEventListener('click', () => {

      const name = document.getElementById('new-name').value.trim();

      const breed = document.getElementById('new-breed').value.trim();

      const birth = document.getElementById('new-birth').value;

      if(!name) return alert('请填写宠物名称');

      const newPet = { id: Date.now(), name: name, breed: breed || "未知", birth: birth || "" , health: [] };

      pets.push(newPet);

      renderPets();

      // 清空输入

      document.getElementById('new-name').value = '';

      document.getElementById('new-breed').value = '';

      document.getElementById('new-birth').value = '';

    });

 

    // 动态健康条目新增

    document.querySelectorAll('.btn-add-health').forEach(btn => {

      btn.addEventListener('click', () => {

        // 在演示中简单地把一条健康记录添加到第一个宠物

        if (pets.length === 0) return;

        const w = document.querySelector('.health-weight')?.value || (Math.random()*2+1).toFixed(1);

        const v = document.querySelector('.health-vax')?.value || "疫苗X" + Math.floor(Math.random()*9+1);

        const d = new Date().toISOString().slice(0,10);

        const rec = { date: d, weight: parseFloat(w), vax: v, dew: "驱虫Z" + Math.floor(Math.random()*9+1) };

        pets[0].health.push(rec);

        // 清空

        document.querySelector('.health-weight').value = '';

        document.querySelector('.health-vax').value = '';

        document.querySelector('.health-dew').value = '';

        renderPets();

        renderTimeline();

      });

    });

    // 发布日常

    publishBtn.addEventListener('click', () => {

      const t = postTitle.value.trim();

      const c = postContent.value.trim();

      const post = { id: Date.now(), author: "你", title: t || "无标题", content: c || "", photo: p || "" };

      const el = document.createElement('div');

      el.className = 'card';

      el.innerHTML = `

        <div style="display:flex; align-items:center; gap:8px;">

          <span class="avatar-sm" aria-label="avatar"></span>

          <strong>你</strong>

          <span class="muted" style="font-size:12px;">刚刚发布</span>

        <div class="muted" style="font-size:12px; margin-top:4px">${post.content}</div>

        ${post.photo ? `<img src="${post.photo}" alt="封面" style="margin-top:6px; border-radius:6px; height:180px; object-fit:cover;" />` : ""}

      `;

      feedListEl.prepend(el);

      // 清空输入

      postTitle.value = '';

      postContent.value = '';
